# Cases zustc does not pass yet: `name  # reason`. A listed case that fails is reported but does not fail
# tests/security.rs; a listed case that passes fails it, so this list only shrinks. `unsafe_api:<name>` lines are rows
# of unsafe_std_apis.txt.

# Borrow and region checking (closures, temporaries, drop check, slices)
borrow_closure_escapes          # E0373 is only checked for thread::spawn: a boxed closure borrowing a local is returned
borrow_closure_mut_and_shared   # a closure's captures are not loans: `v` is read while a closure holds `&mut v`
borrow_slice_into_vec_then_clear # `&v[..]` is not a loan of v: a clear (dropping the elements) is accepted under it
borrow_uninit_read              # rejected, but reported at the function's first line instead of the read
region_temporary_dropped        # E0716 not found when the reference is assigned after its `let`
region_dropck                   # no drop check: a value with Drop may outlive what it borrows

# Privacy (E0603 / E0616 / E0451): visibility is not checked at all
privacy_field
privacy_field_write
privacy_ctor
privacy_fn

# Traits and types
unsafe_fn_pointer               # fn pointer types do not record `unsafe`, so calling one needs no unsafe block
unsafe_nested_fn_in_unsafe_block # rejected with E0133, but at a wrong line (a nested fn's lines counted from its item)
unsafe_union_pattern            # rejected, but as an unsupported pattern (a union pattern of one field), not as E0133
send_dyn_thread                 # `dyn Trait` types do not keep `+ Send`, so a `Box<dyn Fn()>` (maybe holding an Rc) moves to a thread
cve_2019_12083_error_type_id    # an impl may define a method its trait does not declare (E0407); zustc has no downcast, so no type confusion follows
lit_overflow                    # overflowing_literals: `256` as u8 is accepted (and wraps to 0)
const_overflow                  # E0080: overflow in a const is found only when it is evaluated at run time (a panic)

# Runtime
rt_refcell                      # RefCell hands out plain references with no borrow flag: two `borrow_mut` alias
rt_box_move_out                 # `let x = *b;` moves the value but never frees the Box's block (a leak)
rt_box_move_patterns            # the same leak through patterns (`let Pair { a, b } = *b`, `match *b`); no double free
